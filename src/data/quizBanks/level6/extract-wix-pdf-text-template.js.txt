async function() {
  const sources = __SOURCES__;
  const zipRequest = await wix.request({
    scope: "site", method: "POST",
    url: "https://www.wixapis.com/site-media/v1/files/generate-download-url",
    body: { fileIds: sources.map(s => s.wixId) }
  });
  const zip = await wix.request({
    scope: "account", method: "GET",
    url: zipRequest.data.downloadUrl, responseType: "base64"
  });
  const entries = await wix.unzip(zip.data);
  function latin(bytes) {
    let s = "";
    for (let i = 0; i < bytes.length; i += 8192)
      s += String.fromCharCode(...bytes.subarray(i, Math.min(i + 8192, bytes.length)));
    return s;
  }
  function unescapePdf(t) {
    return t.replace(/\\([0-7]{1,3}|.)/g, (_, v) => {
      if (/^[0-7]/.test(v)) return String.fromCharCode(parseInt(v, 8));
      return ({n: "\n", r: "\r", t: "\t", b: "\b", f: "\f"})[v] ?? v;
    });
  }
  function extractText(source) {
    const segments = [];
    for (const block of source.matchAll(/\bBT([\s\S]*?)\bET/g)) {
      let tokens = [...block[1].matchAll(/\(((?:\\.|[^\\)])*)\)/g)]
        .map(m => unescapePdf(m[1]));
      if (!tokens.length) tokens = [...block[1].matchAll(/<([0-9a-fA-F]{4,})>/g)].map(m => {
        let result = "";
        for (let i = 0; i < m[1].length; i += 4)
          result += String.fromCharCode(parseInt(m[1].slice(i, i + 4), 16));
        return result;
      });
      if (tokens.length) segments.push(tokens.join(""));
    }
    return segments.join(" | ").replace(/\u0000/g, "").slice(0, 18000);
  }
  const outputs = [];
  for (const item of sources) {
    const entry = entries.find(e => e.path === item.sourcePath.split("/").at(-1));
    if (!entry) { outputs.push({projectId: item.projectId, error: "Archive entry missing"}); continue; }
    const b = entry.bytes;
    const raw = latin(b);
    const markers = [...raw.matchAll(/(?:^|\r?\n)(\d+)\s+\d+\s+obj\b/g)];
    const objects = new Map();
    for (let i = 0; i < markers.length; i++) {
      objects.set(+markers[i][1], {
        offset: markers[i].index,
        text: raw.slice(markers[i].index, markers[i + 1]?.index ?? raw.length)
      });
    }
    const catalog = [...objects.values()].find(x => /\/Type\s*\/Catalog\b/.test(x.text));
    const root = +(catalog?.text.match(/\/Pages\s+(\d+)\s+\d+\s+R/) || [])[1];
    const pages = [];
    function gather(id, depth) {
      if (depth > 10 || !objects.has(id)) return;
      const t = objects.get(id).text;
      if (/\/Type\s*\/Page\b/.test(t)) { pages.push(id); return; }
      const kids = t.match(/\/Kids\s*\[([^\]]+)\]/);
      if (kids) for (const match of kids[1].matchAll(/(\d+)\s+\d+\s+R/g))
        gather(+match[1], depth + 1);
    }
    if (root) gather(root, 0);
    if (!pages.length) for (const [id, v] of objects)
      if (/\/Type\s*\/Page\b/.test(v.text)) pages.push(id);
    const outputPages = [];
    for (let p = 0; p < pages.length; p++) {
      const descriptor = objects.get(pages[p]).text;
      const single = descriptor.match(/\/Contents\s+(\d+)\s+\d+\s+R/);
      const array = descriptor.match(/\/Contents\s*\[([^\]]+)\]/);
      const ids = single ? [+single[1]] : array
        ? [...array[1].matchAll(/(\d+)\s+\d+\s+R/g)].map(m => +m[1])
        : [];
      const chunks = [];
      for (const id of ids) {
        const stream = objects.get(id);
        if (!stream) continue;
        const marker = stream.text.match(/stream\r?\n/);
        if (!marker) continue;
        const length = +(stream.text.match(/\/Length\s+(\d+)/) || [])[1];
        if (!length) continue;
        const start = stream.offset + marker.index + marker[0].length;
        const bytes = b.subarray(start, start + length);
        let uncompressed = bytes;
        if (/\/FlateDecode/.test(stream.text.slice(0, marker.index))) {
          try {
            uncompressed = new Uint8Array(await new Response(
              new Blob([bytes]).stream().pipeThrough(new DecompressionStream("deflate"))
            ).arrayBuffer());
          } catch (error) {
            chunks.push("[decompression error: " + String(error) + "]");
            continue;
          }
        }
        chunks.push(extractText(latin(uncompressed)));
      }
      outputPages.push({slide: p + 1, text: chunks.join(" ").slice(0, 18000)});
    }
    outputs.push({
      projectId: item.projectId, sourcePath: item.sourcePath,
      wixFileId: item.wixId, sourceBytes: b.length,
      pageCount: pages.length, pages: outputPages,
      extractionMethod: "original Wix Media Manager PDF binary, ordered PDF Page tree, zlib FlateDecode, text operators",
      visualReviewComplete: false
    });
  }
  return outputs;
}
